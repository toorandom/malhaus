REST API & MCP

Submit files and URLs for automated malware analysis. Results appear in the web UI alongside browser submissions. To obtain an API key, contact the administrator of this malhaus instance — keys are issued via the admin dashboard and may carry a rate limit and expiry date.

Authentication

Pass your API key as a bearer token in the Authorization header on every request:

Authorization: Bearer mh_xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx

Keys are issued by the admin and may carry a per-hour rate limit and an optional expiry date. Revoked or expired keys receive a 401 Unauthorized response.

Python examples

Copy the script below and run it directly. Requires the requests library — pip install requests.

#!/usr/bin/env python3
"""
malhaus-submit — submit files or URLs to a malhaus instance via the REST API.

Usage
-----
  # Analyze a file
  python malhaus_submit.py --file suspicious.exe

  # Analyze with Ghidra decompilation (deeper, slower)
  python malhaus_submit.py --file binary.exe --ghidra

  # Submit a URL
  python malhaus_submit.py --url https://example.com/payload.ps1

  # Submit an encrypted archive (ZIP / 7z / RAR)
  python malhaus_submit.py --file malware.zip --password infected

  # Override server / key for this run
  python malhaus_submit.py --file sample.exe --server https://malhaus.example.com --key mh_xxx

Environment variables (used when --server / --key are not passed):
  MALHAUS_URL   base URL of the malhaus instance
  MALHAUS_KEY   your API key  (mh_…)
"""

import argparse
import os
import sys
import time
import requests

# ── Default configuration ──────────────────────────────────────────────────────
DEFAULT_SERVER = os.environ.get("MALHAUS_URL", "https://your-malhaus-instance.example.com")
DEFAULT_KEY    = os.environ.get("MALHAUS_KEY",  "mh_xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx")
POLL_INTERVAL  = 5   # seconds between status checks


# ── Helpers ────────────────────────────────────────────────────────────────────
def submit(server: str, key: str, args: argparse.Namespace) -> str:
    """Submit a file or URL and return the job_id."""
    headers = {"Authorization": f"Bearer {key}"}
    endpoint = f"{server}/api/v1/analyze"

    if args.file:
        path = args.file
        extra = {}
        if args.password:
            extra["archive_password"] = args.password   # encrypted ZIP / 7z / RAR
        if args.ghidra:
            extra["use_ghidra"] = "1"                   # Ghidra decompilation (takes longer)

        with open(path, "rb") as fh:
            resp = requests.post(
                endpoint,
                headers=headers,
                files={"file": (os.path.basename(path), fh, "application/octet-stream")},
                data=extra,
                timeout=120,
            )
    elif args.url:
        extra = {}
        if args.ghidra:
            extra["use_ghidra"] = "1"
        resp = requests.post(
            endpoint,
            headers=headers,
            data={"url": args.url, **extra},
            timeout=120,
        )
    else:
        print("error: provide --file or --url", file=sys.stderr)
        sys.exit(1)

    if resp.status_code == 401:
        print("error: API key rejected (401 Unauthorized)", file=sys.stderr)
        sys.exit(1)
    if resp.status_code == 429:
        print("error: rate limit exceeded (429). Try again later.", file=sys.stderr)
        sys.exit(1)
    resp.raise_for_status()
    return resp.json()["job_id"]


def poll(server: str, key: str, job_id: str) -> dict:
    """Poll until the job is done or failed. Returns the result dict."""
    headers = {"Authorization": f"Bearer {key}"}
    url = f"{server}/api/v1/jobs/{job_id}"
    while True:
        resp = requests.get(url, headers=headers, timeout=30)
        resp.raise_for_status()
        data = resp.json()
        status = data.get("status", "unknown")
        print(f"  [{status}]", flush=True)
        if status in ("done", "failed"):
            return data
        time.sleep(POLL_INTERVAL)


def print_result(data: dict) -> None:
    """Print a human-readable summary of a completed job."""
    if data.get("status") == "failed":
        print("Analysis failed:", data.get("error"))
        return

    verdict    = data.get("verdict", {})
    risk       = verdict.get("risk_level", "unknown")
    confidence = verdict.get("confidence", "?")
    file_type  = verdict.get("file_type", "")
    reasons    = data.get("top_reasons", [])
    tools      = data.get("tools_used", [])
    sha256     = data.get("sha256", "")
    h_score    = data.get("heuristic_score", "?")
    s_score    = data.get("strings_score", "?")
    report_url = data.get("report_url", "")

    print(f"\n{'─'*60}")
    print(f"  risk           : {risk.upper()}")
    print(f"  confidence     : {confidence}%")
    print(f"  file type      : {file_type}")
    print(f"  heuristic score: {h_score}/100")
    print(f"  strings score  : {s_score}/100")
    if sha256:
        print(f"  sha256         : {sha256}")
    if report_url:
        print(f"  report         : {report_url}")
    print(f"\n  Top reasons:")
    for reason in reasons[:5]:
        print(f"    • {reason}")
    if tools:
        print(f"\n  Tools used : {', '.join(tools)}")
    print(f"{'─'*60}\n")


# ── Entry point ────────────────────────────────────────────────────────────────
def main():
    parser = argparse.ArgumentParser(
        prog="malhaus_submit",
        description="Submit a file or URL to malhaus for static malware analysis.",
    )
    src = parser.add_mutually_exclusive_group(required=True)
    src.add_argument("--file",     metavar="PATH", help="Path to the file to analyse")
    src.add_argument("--url",      metavar="URL",  help="URL to download and analyse")
    parser.add_argument("--password", metavar="PW",  default="",
                        help="Password for encrypted ZIP / 7z / RAR archives")
    parser.add_argument("--ghidra", action="store_true",
                        help="Enable Ghidra decompilation (deeper analysis, takes longer)")
    parser.add_argument("--server", metavar="URL",  default=DEFAULT_SERVER,
                        help="malhaus base URL (default: $MALHAUS_URL or hard-coded above)")
    parser.add_argument("--key",    metavar="KEY",  default=DEFAULT_KEY,
                        help="API key (default: $MALHAUS_KEY or hard-coded above)")
    args = parser.parse_args()

    target = args.file or args.url
    print(f"Submitting: {target}")
    job_id = submit(args.server, args.key, args)
    print(f"Job ID: {job_id}")
    print("Polling for results…")
    result = poll(args.server, args.key, job_id)
    print_result(result)


if __name__ == "__main__":
    main()

MCP (Model Context Protocol)

Malhaus exposes an MCP server so AI agents (Claude Code, Claude Desktop, OpenCode, Cursor, custom Python agents…) can call malware analysis as a native tool — no shell scripts, no manual polling. Connect any MCP client to:

https://your-malhaus-instance.com/mcp

For server setup and client configuration see README-MCP.md.

Available tools

ToolArgumentsDescription
analyze_file file_b64, filename Analyze a file from base64-encoded content. Returns verdict, confidence, heuristic score, top reasons, IOCs. Typical duration: 30–120 s.
analyze_url url Download a file from a URL and analyze it.
analyze_sha256 sha256 Retrieve a cached result by SHA-256 hash. Instant if previously analyzed.

How it works

Local file (e.g. /home/analyst/samples/dropper.exe or C:\Users\analyst\Downloads\invoice.exe)
  → your code reads the file and base64-encodes it
  → calls analyze_file(file_b64="TVqQAAMA...", filename="dropper.exe")
  → server decodes, runs analysis, returns verdict

Remote URL (e.g. https://cdn.example.com/payload.ps1)
  → calls analyze_url(url="https://cdn.example.com/payload.ps1")
  → server downloads and analyzes it directly

Example: analyze a URL

Install: pip install mcp httpx

#!/usr/bin/env python3
"""
Analyze a file by URL using the malhaus MCP server.
Usage: python analyze_url.py https://cdn.example.com/suspicious.exe

pip install mcp httpx
"""
import asyncio, sys, httpx
from mcp import ClientSession
from mcp.client.streamable_http import streamablehttp_client

SERVER_URL = "https://your-malhaus-instance.com/mcp"

# Note: pass httpx_client_factory=lambda **kw: httpx.AsyncClient(verify=False, **kw)
# if your server uses a self-signed certificate.

async def main(url: str):
    async with streamablehttp_client(SERVER_URL) as (r, w, _):
        async with ClientSession(r, w) as s:
            await s.initialize()
            result = await s.call_tool("analyze_url", {"url": url})
            print(result.content[0].text)

asyncio.run(main(sys.argv[1]))

Example: analyze a local file

The file is read on your machine, base64-encoded, and sent to the server. The server never accesses your filesystem directly. 

#!/usr/bin/env python3
"""
Analyze a local file using the malhaus MCP server.
Usage: python analyze_file.py /path/to/suspicious.exe
       python analyze_file.py "C:\\Users\\analyst\\Downloads\\invoice.exe"

pip install mcp httpx
"""
import asyncio, base64, os, sys, httpx
from mcp import ClientSession
from mcp.client.streamable_http import streamablehttp_client

SERVER_URL = "https://your-malhaus-instance.com/mcp"

# Note: pass httpx_client_factory=lambda **kw: httpx.AsyncClient(verify=False, **kw)
# if your server uses a self-signed certificate.

async def main(path: str):
    with open(path, "rb") as fh:
        file_b64 = base64.b64encode(fh.read()).decode()
    filename = os.path.basename(path)
    print(f"Submitting: {filename} ({os.path.getsize(path):,} bytes)", flush=True)
    async with streamablehttp_client(SERVER_URL) as (r, w, _):
        async with ClientSession(r, w) as s:
            await s.initialize()
            result = await s.call_tool(
                "analyze_file",
                {"file_b64": file_b64, "filename": filename},
            )
            print(result.content[0].text)

asyncio.run(main(sys.argv[1]))

Endpoints

Method Path Description
POST /api/v1/analyze Submit a file or URL for analysis. Returns {"job_id": "..."} immediately. Body can be multipart/form-data (file upload) or application/x-www-form-urlencoded (URL submission). Optional fields: use_ghidra=1, archive_password.
GET /api/v1/jobs/<job_id> Poll job status. Returns a JSON object with status (pending / running / done / error), and on completion a result dict or error string.

Response fields (status: done)

  • status pending · running · done · failed — current job state. Poll until done or failed.
  • job_id UUID of the analysis job, as returned by POST /api/v1/analyze.
  • verdict.risk_level benign · suspicious · likely_malware — overall verdict from the LLM loop.
  • verdict.confidence Integer 0–100 — how confident the model is in the verdict.
  • verdict.file_type Detected file type string (e.g. pe, elf, pdf).
  • top_reasons List of strings — the most significant evidence items from the analysis.
  • heuristic_score Deterministic score 0–100 computed before the LLM pass (entropy, imports, IOC hits, etc.).
  • strings_score Score 0–100 based on string analysis (suspicious patterns, URLs, registry keys, etc.).
  • strings_summary Short LLM-generated summary of notable strings found in the file.
  • tools_used List of static analysis tools that ran (e.g. objdump, olevba, ghidra).
  • tool_outputs Dict mapping tool name → {stdout, stderr, error} — raw output from each tool.
  • sha256 SHA-256 of the submitted file. Results are cached by hash — resubmitting an identical file returns instantly.
  • report_url Relative URL to the HTML report page for this analysis.
  • report_json_url Relative URL to the full raw JSON report.
  • html_summary Pre-rendered HTML snippet with a human-readable summary — useful for embedding in ticketing systems or chat bots.
